const { onCall, HttpsError } = require("firebase-functions/v2/https"); const { logger } = require("firebase-functions"); const { initializeApp } = require("firebase-admin/app"); const { getFirestore } = require("firebase-admin/firestore"); const { getStorage } = require("firebase-admin/storage"); initializeApp(); exports.unlockGallery = onCall({ region: "northamerica-northeast2" }, async (request) => { try { const code = String(request.data.code || "").trim().toUpperCase(); if (!code) { throw new HttpsError("invalid-argument", "Missing access code."); } const db = getFirestore(); const snap = await db.collection("accessCodes").doc(code).get(); if (!snap.exists) { throw new HttpsError("permission-denied", "Invalid access code."); } const data = snap.data(); if (!data.active) { throw new HttpsError("permission-denied", "Gallery inactive."); } const bucket = getStorage().bucket(); const total = parseInt(data.count, 10); logger.info("COUNT DEBUG", { count: data.count, parsed: total, type: typeof data.count, folder: data.folder }); if (isNaN(total) || total < 1) { throw new HttpsError("internal", "Invalid image count."); } const expires = Date.now() + 60 * 60 * 1000; const images = []; for (let i = 1; i <= total; i++) { const filePath = `galleries/${data.folder}/${i}.png`; const file = bucket.file(filePath); logger.info("FILE DEBUG", { filePath }); const [url] = await file.getSignedUrl({ version: "v4", action: "read", expires }); images.push(url); } logger.info("RETURN DEBUG", { sessionTitle: data.sessionTitle, imageCount: images.length }); return { sessionTitle: data.sessionTitle, images }; } catch (error) { logger.error("unlockGallery failed", error); throw new HttpsError("internal", error.message || "Internal error"); } });
